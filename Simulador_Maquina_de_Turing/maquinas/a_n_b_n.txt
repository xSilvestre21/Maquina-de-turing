# Reconhece a linguagem L = { a^n b^n | n >= 0 }, que nao e regular.
#
# Estrategia: marca o 'a' mais a esquerda com X, procura o 'b' mais a esquerda
# e o marca com Y, volta ao inicio e repete. No fim, so pode restar Y na fita.
#
# Aceita: (vazia), ab, aabb, aaabbb
# Rejeita: aab, abb, ba, aba

NOME: a^n b^n
INICIAL: q0
FINAIS: qf

TRANSICOES:
# q0 - procura o proximo 'a' nao marcado
q0 a q1 X R
q0 Y q3 Y R
q0 _ qf _ R

# q1 - anda para a direita ate achar o primeiro 'b' nao marcado
q1 a q1 a R
q1 Y q1 Y R
q1 b q2 Y L

# q2 - volta para a esquerda ate o ultimo 'a' ja marcado
q2 a q2 a L
q2 Y q2 Y L
q2 X q0 X R

# q3 - confere se sobraram apenas 'b' marcados (Y) ate o fim da fita
q3 Y q3 Y R
q3 _ qf _ R
