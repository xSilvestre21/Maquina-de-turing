# Complemento de 1 de um numero binario (exemplo de validacao da tarefa).
# Inverte cada 0 para 1 e cada 1 para 0 e para no final da fita.
#
# Alfabeto de entrada (E): {0, 1}
# Alfabeto da fita (G)....: {0, 1, _}
#
# Teste sugerido: palavra 101  ->  fita final 010 (PALAVRA ACEITA em qf)

NOME: Complemento de 1 (inverte bits)
INICIAL: q0
FINAIS: qf

TRANSICOES:
# [estado] [lido] [novo_estado] [escrito] [direcao]
